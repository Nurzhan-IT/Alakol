3.2. JavaScript для улучшения UX:
```javascript
// static/js/legal.js
document.addEventListener('DOMContentLoaded', function() {
    // Автоматическое открытие документов в новых вкладках
    const legalLinks = document.querySelectorAll('a[href*="/legal/"]');
    legalLinks.forEach(link => {
        if (link.getAttribute('target') !== '_blank') {
            link.setAttribute('target', '_blank');
            link.setAttribute('rel', 'noopener noreferrer');
        }
    });
    
    // Подсветка обязательных чекбоксов
    const requiredCheckboxes = document.querySelectorAll('input[type="checkbox"][required]');
    requiredCheckboxes.forEach(checkbox => {
        const label = document.querySelector(`label[for="${checkbox.id}"]`);
        if (label) {
            label.innerHTML += ' <span class="text-danger">*</span>';
        }
    });
    
    // Валидация согласий перед отправкой формы
    const forms = document.querySelectorAll('form');
    forms.forEach(form => {
        form.addEventListener('submit', function(e) {
            const requiredConsents = form.querySelectorAll('input[type="checkbox"][required]');
            let allChecked = true;
            
            requiredConsents.forEach(checkbox => {
                if (!checkbox.checked) {
                    allChecked = false;
                    checkbox.closest('.form-check').style.border = '2px solid #dc3545';
                }
            });
            
            if (!allChecked) {
                e.preventDefault();
                alert('Пожалуйста, подтвердите все обязательные согласия');
            }
        });
    });
});
```

```python
# userauths/models.py (дополнение)
class UserConsent(models.Model):
    CONSENT_TYPES = (
        ('terms_of_use', 'Пользовательское соглашение'),
        ('privacy_policy', 'Политика конфиденциальности'),
        ('personal_data', 'Обработка персональных данных'),
        ('marketing', 'Маркетинговые сообщения'),
        ('booking_terms', 'Правила бронирования'),
        ('payment_terms', 'Правила оплаты'),
    )
    
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    consent_type = models.CharField(max_length=50, choices=CONSENT_TYPES)
    given_at = models.DateTimeField(auto_now_add=True)
    ip_address = models.GenericIPAddressField()
    user_agent = models.TextField()
    document_version = models.CharField(max_length=10, default='1.0')
    is_active = models.BooleanField(default=True)
    withdrawn_at = models.DateTimeField(null=True, blank=True)
    
    class Meta:
        unique_together = ['user', 'consent_type', 'document_version']
        
    def __str__(self):
        return f"{self.user} - {self.consent_type} - {self.document_version}"
```

4.2. Функция для сохранения согласий:
```python
# userauths/utils.py
def save_user_consent(user, consent_type, request, document_version='1.0'):
    """Сохраняет согласие пользователя"""
    UserConsent.objects.update_or_create(
        user=user,
        consent_type=consent_type,
        document_version=document_version,
        defaults={
            'ip_address': get_client_ip(request),
            'user_agent': request.META.get('HTTP_USER_AGENT', ''),
            'is_active': True,
            'withdrawn_at': None
        }
    )

def get_client_ip(request):
    """Получает IP-адрес клиента"""
    x_forwarded_for = request.META.get('HTTP_X_FORWARDED_FOR')
    if x_forwarded_for:
        ip = x_forwarded_for.split(',')[0]
    else:
        ip = request.META.get('REMOTE_ADDR')
    return ip
```

7. УПРАВЛЕНИЕ ВЕРСИЯМИ ДОКУМЕНТОВ

7.1. Модель для версионирования:
```python
# legal/models.py
class LegalDocument(models.Model):
    DOCUMENT_TYPES = (
        ('terms_of_use', 'Пользовательское соглашение'),
        ('privacy_policy', 'Политика конфиденциальности'),
        ('public_offer', 'Публичная оферта'),
        ('booking_rules', 'Правила бронирования'),
        ('personal_data_consent', 'Согласие на обработку персональных данных'),
        ('payment_rules', 'Правила оплаты'),
    )
    
    document_type = models.CharField(max_length=50, choices=DOCUMENT_TYPES)
    version = models.CharField(max_length=10)
    content = models.TextField()
    effective_date = models.DateTimeField()
    created_at = models.DateTimeField(auto_now_add=True)
    is_active = models.BooleanField(default=False)
    language = models.CharField(max_length=2, default='ru')
    
    class Meta:
        unique_together = ['document_type', 'version', 'language']
        
    def save(self, *args, **kwargs):
        if self.is_active:
            # Деактивировать предыдущие версии
            LegalDocument.objects.filter(
                document_type=self.document_type,
                language=self.language,
                is_active=True
            ).update(is_active=False)
        super().save(*args, **kwargs)
```

```python
class LegalDocumentView(TemplateView):

        # Записываем просмотр документа

            document_type=self.document_type,
            user=request.user if request.user.is_authenticated else None,
```
